# GoAWD 架构设计文档

## 1. 整体架构

GoAWD 采用分层架构设计，主要分为以下层次：

```
┌─────────────────────────────────────────────────────────┐
│                   表现层 (Presentation)                  │
│            Frontend (Vue.js) / CLI Tools                 │
└─────────────────────────────────────────────────────────┘
                           │
┌─────────────────────────────────────────────────────────┐
│                   接口层 (API Layer)                     │
│         HTTP Server / WebSocket / TCP Receiver           │
└─────────────────────────────────────────────────────────┘
                           │
┌─────────────────────────────────────────────────────────┐
│                   业务层 (Business Logic)                │
│     Plugin Manager / Alert Engine / Data Processor       │
└─────────────────────────────────────────────────────────┘
                           │
┌─────────────────────────────────────────────────────────┐
│                   数据层 (Data Layer)                    │
│           MongoDB Driver / Cache / File Storage          │
└─────────────────────────────────────────────────────────┘
```

## 3. 核心模块设计

### 3.1 Server 模块

```go
// internal/core/server.go
type Server struct {
    config     *config.Config
    httpServer *http.Server
    receiver   *Receiver
    db         *Database
    hub        *Hub
    pluginMgr  *plugin.Manager
}

func NewServer(cfg *config.Config) *Server {
    return &Server{
        config:   cfg,
        db:       NewDatabase(cfg.MongoURI),
        hub:      NewHub(),
        receiver: NewReceiver(cfg.TCPPort, NewDatabase(cfg.MongoURI)),
        pluginMgr: plugin.NewManager(),
    }
}

func (s *Server) Run(ctx context.Context) error {
    g, ctx := errgroup.WithContext(ctx)
    
    // 启动 HTTP Server
    g.Go(func() error {
        return s.httpServer.ListenAndServe()
    })
    
    // 启动 TCP Receiver
    g.Go(func() error {
        return s.receiver.Start(ctx)
    })
    
    // 启动 WebSocket Hub
    g.Go(func() error {
        return s.hub.Run(ctx)
    })
    
    // 启动定时任务
    g.Go(func() error {
        return s.startScheduler(ctx)
    })
    
    return g.Wait()
}
```

### 3.2 Receiver 模块

```go
// internal/core/receiver.go
type Receiver struct {
    listener   net.Listener
    db         *Database
    pluginMgr  *plugin.Manager
    hub        *Hub
    
    // PWN 会话管理
    pwnSockets sync.Map  // connID -> *PwnSocket
    pwnProcess sync.Map  // pid -> *PwnProcess
    
    // 进程快照
    currentProcs sync.Map // pid -> *ProcessInfo
}

func (r *Receiver) Start(ctx context.Context) error {
    listener, err := net.Listen("tcp", fmt.Sprintf(":%d", r.port))
    if err != nil {
        return err
    }
    defer listener.Close()
    
    for {
        conn, err := listener.Accept()
        if err != nil {
            select {
            case <-ctx.Done():
                return ctx.Err()
            default:
                log.Printf("accept error: %v", err)
                continue
            }
        }
        go r.handleConn(ctx, conn)
    }
}

func (r *Receiver) handleConn(ctx context.Context, conn net.Conn) {
    defer conn.Close()
    
    scanner := bufio.NewScanner(conn)
    for scanner.Scan() {
        select {
        case <-ctx.Done():
            return
        default:
            line := scanner.Bytes()
            r.processMessage(line, conn)
        }
    }
}
```

```go
// internal/core/database.go
type Database struct {
    client   *mongo.Client
    database *mongo.Database
    
    // 计数缓存
    countCache sync.Map
}

// MongoDB Collections
const (
    CollWeb         = "web"
    CollPWN         = "pwn"
    CollFilesystem  = "filesystem"
    CollProcess     = "process"
    CollAlert       = "alert"
)

func (d *Database) InsertOne(ctx context.Context, collection string, data interface{}) (primitive.ObjectID, error) {
    // 转义键名中的特殊字符
    escaped := EscapeKeys(data)
    
    result, err := d.database.Collection(collection).InsertOne(ctx, escaped)
    if err != nil {
        return primitive.NilObjectID, err
    }
    
    // 更新计数缓计
    d.incrementCount(collection)
    
    return result.InsertedID.(primitive.ObjectID), nil
}

func (d *Database) FindWithPagination(ctx context.Context, collection string, page, count int, projection interface{}) (*PaginatedResult, error) {
    total := d.GetCollectionCount(collection)
    totalPages := int(math.Ceil(float64(total) / float64(count)))
    
    offset := (page - 1) * count
    if page == 0 {
        offset = (totalPages - 1) * count
    }
    
    cursor, err := d.database.Collection(collection).Find(ctx, bson.M{}, options.Find().
        SetProjection(projection).
        SetSkip(int64(offset)).
        SetLimit(int64(count)))
    if err != nil {
        return nil, err
    }
    
    var results []bson.M
    if err := cursor.All(ctx, &results); err != nil {
        return nil, err
    }
    
    return &PaginatedResult{
        Page:     page,
        LastPage: totalPages,
        Data:     results,
    }, nil
}
```

### 3.4 Plugin 模块

```go
// internal/plugin/manager.go
type Manager struct {
    hooks    map[string][]HookFunc
    plugins  []Plugin
    caller   interface{}
    mu       sync.RWMutex
}

type HookFunc func(data interface{}) interface{}

type Plugin interface {
    Name() string
    Register(m *Manager)
}

func NewManager() *Manager {
    return &Manager{
        hooks: make(map[string][]HookFunc),
    }
}

func (m *Manager) Register(routine, operation string, hook HookFunc) {
    m.mu.Lock()
    defer m.mu.Unlock()
    
    key := strings.ToLower(routine + "/" + operation)
    m.hooks[key] = append(m.hooks[key], hook)
}

func (m *Manager) Invoke(caller interface{}, routine, operation string, data interface{}) interface{} {
    m.mu.RLock()
    defer m.mu.RUnlock()
    
    key := strings.ToLower(routine + "/" + operation)
    hooks, ok := m.hooks[key]
    if !ok {
        return data
    }
    
    m.caller = caller
    for _, hook := range hooks {
        data = hook(data)
    }
    m.caller = nil
    
    return data
}

func (m *Manager) GetCaller() interface{} {
    return m.caller
}
```

## 4. 数据流设计

### 4.1 Web 日志处理流程

```
探针 ──TCP──> Receiver ──> handleWeb()
                           │
                           ├── urldecode 数据
                           ├── 调用 PluginManager.Invoke("Web", "processLog", data)
                           │           │
                           │           ├── FlagBuster.processWebBuffer(data)
                           │           │   └── 替换 flag 字段，产生告警
                           │           └── 返回处理后的 data
                           │
                           ├── 发送响应给探针（篡改后的 buffer）
                           └── 插入 MongoDB "web" 集合
                                    │
                                    └── WebSocket Hub 广播 "web" 更新
```

### 4.2 PWN 日志处理流程

```
Guardian ──TCP──> Receiver ──> handleNewPWN()
                               │
                               ├── 记录 PwnSocket 和 PwnProcess
                               └── 切换到流模式 (useStream)
                                    │
                                    ▼
                           processPWNStream()
                               │
                               ├── 按 pid 合并 stdin/stdout 流
                               ├── 检测连接关闭 (data === false)
                               │           │
                               │           ├── 调用 PluginManager.Invoke("PWN", "processLog", data)
                               │           ├── Base64 编码 streamlog
                               │           └── 插入 MongoDB "pwn" 集合
                               │
                               └── WebSocket Hub 广播 "pwn" 更新
```

```
RoundWorm ──TCP──> Receiver ──> handleNewFile()

                               ├── 解析 inotify 事件类型
```

## 5. 错误处理策略

### 5.1 优雅关闭

```go
func (s *Server) Shutdown(ctx context.Context) error {
    // 1. 停止接受新连接
    s.httpServer.Shutdown(ctx)
    
    // 2. 等待现有请求完成
    <-ctx.Done()
    
    // 3. 关闭 TCP 监听
    s.receiver.Stop()
    
    // 4. 刷新 MongoDB 缓冲区
    s.db.Flush(ctx)
    
    // 5. 关闭数据库连接
    s.db.Close()
    
    return nil
}
```

### 5.2 错误恢复

```go
func (r *Receiver) handleConnWithRecovery(conn net.Conn) {
    defer func() {
        if err := recover(); err != nil {
            log.Printf("panic recovered: %v\n%s", err, debug.Stack())
        }
        conn.Close()
    }()
    
    r.handleConn(conn)
}
```

## 7. 安全设计

### 7.1 Token 鉴权

```go
func TokenAuthMiddleware(token string) func(http.Handler) http.Handler {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            // 从 Query 或 Header 获取 token
            requestToken := r.URL.Query().Get("token")
            if requestToken == "" {
                requestToken = r.Header.Get("Token")
            }
            
            if token != "" && requestToken != token {
                http.Error(w, "Forbidden", http.StatusForbidden)
                return
            }
            
            next.ServeHTTP(w, r)
        })
    }
}
```

### 7.2 数据过滤

```go
// 转义 MongoDB 键名中的特殊字符
func EscapeKeys(data interface{}) interface{} {
    switch v := data.(type) {
    case map[string]interface{}:
        result := make(map[string]interface{})
        for key, value := range v {
            // 替换 $ 和 . 为全角字符
            escapedKey := strings.ReplaceAll(key, "$", "＄")
            escapedKey = strings.ReplaceAll(escapedKey, ".", "．")
            result[escapedKey] = EscapeKeys(value)
        }
        return result
    case []interface{}:
        result := make([]interface{}, len(v))
        for i, item := range v {
            result[i] = EscapeKeys(item)
        }
        return result
    default:
        return v
    }
}
```
