# GoAWD 数据结构定义

本文档定义 GoAWD 中所有核心数据结构的 Go 语言实现。

## 1. 基础类型定义

```go
package types

import (
    "time"
    "go.mongodb.org/mongo-driver/bson/primitive"
)
```

## 2. TCP 探针协议数据结构

### 2.1 消息封装

```go
// ProbeMessage 探针消息封装
type ProbeMessage struct {
    Type string      `json:"type"`
    Data interface{} `json:"data"`
}

// 消息类型常量
const (
    MsgTypeWeb            = "web"
    MsgTypeNewProcess     = "new_process"
    MsgTypeCurrentProcess = "pid_list"
    MsgTypeNewFile        = "file"
    MsgTypePWN            = "pwn"
    MsgTypePing           = "ping"
)
```

### 2.2 Web 流量数据

```go
// WebLogData Web流量日志数据
type WebLogData struct {
    Script string            `json:"script" bson:"script"`
    Method string            `json:"method" bson:"method"`
    URI    string            `json:"uri" bson:"uri"`
    Remote string            `json:"remote" bson:"remote"`
    Header map[string]string `json:"header" bson:"header"`
    GET    map[string]string `json:"get" bson:"get"`
    POST   map[string]string `json:"post" bson:"post"`
    Cookie map[string]string `json:"cookie" bson:"cookie"`
    File   []FileUpload      `json:"file" bson:"file"`
    Buffer string            `json:"buffer" bson:"buffer"`
}

// FileUpload 文件上传信息
type FileUpload struct {
    Name     string `json:"name" bson:"name"`
    Type     string `json:"type" bson:"type"`
    Size     int64  `json:"size" bson:"size"`
    TempName string `json:"tmp_name" bson:"tmp_name"`
}
```

### 2.3 PWN 流量数据

```go
// PWNInitData PWN初始化数据
type PWNInitData struct {
    File string `json:"file" bson:"file"`
    Type string `json:"type" bson:"type"` // "stdin" or "stdout"
    PID  int    `json:"pid" bson:"pid"`
    Maps string `json:"maps" bson:"maps"` // base64 encoded
}

// PwnSocket PWN套接字信息
type PwnSocket struct {
    ConnID int    `bson:"-"`
    PID    int    `bson:"pid"`
    Type   string `bson:"type"` // "stdin" or "stdout"
}

// PwnProcess PWN进程信息
type PwnProcess struct {
    Time      int64          `json:"time" bson:"time"`
    Bin       string         `json:"bin" bson:"bin"`
    Maps      string         `json:"maps" bson:"maps"`
    Stdin     StreamStat     `json:"stdin" bson:"stdin"`
    Stdout    StreamStat     `json:"stdout" bson:"stdout"`
    StreamLog []StreamLog    `json:"streamlog" bson:"streamlog"`
}

// StreamStat 流统计
type StreamStat struct {
    Group int `json:"group" bson:"group"`
    Byte  int `json:"byte" bson:"byte"`
}

// StreamLog 流日志
type StreamLog struct {
    Type   string `json:"type" bson:"type"`     // "stdin" or "stdout"
    Buffer string `json:"buffer" bson:"buffer"` // base64 encoded
}
```

### 2.4 文件系统事件数据

```go
// FileEventData 文件系统事件数据
type FileEventData struct {
    Time    int64  `json:"time" bson:"time"`
    Path    string `json:"path" bson:"path"`
    IsDir   bool   `json:"isdir" bson:"isdir"`
    Oper    string `json:"oper" bson:"oper"`     // ACCESS, MODIFY, CREATE, DELETE etc.
    Size    int64  `json:"size" bson:"size"`
    Content string `json:"content" bson:"content"` // base64 encoded
}

// inotify事件类型映射
var InotifyEventMap = map[uint32]string{
    0x00000001: "ACCESS",
    0x00000002: "MODIFY",
    0x00000004: "ATTRIB",
    0x00000008: "CLOSE_WRITE",
    0x00000010: "CLOSE_NOWRITE",
    0x00000018: "CLOSE",
    0x00000020: "OPEN",
    0x00000040: "MOVED_FROM",
    0x00000080: "MOVED_TO",
    0x000000C0: "MOVE",
    0x00000100: "CREATE",
    0x00000200: "DELETE",
    0x00000400: "DELETE_SELF",
    0x00000800: "MOVE_SELF",
}
```

### 3.5 告警日志文档

```go
// AlertLogDocument MongoDB alert集合文档
type AlertLogDocument struct {
    ID        primitive.ObjectID `bson:"_id,omitempty"`
    Time      int64              `bson:"time"`
    Type      string             `bson:"type"`      // Web, PWN, FileSystem, Process
    Plugin    string             `bson:"plugin"`    // 插件名称
    Message   string             `bson:"message"`   // 告警消息
    Reference AlertReference     `bson:"reference"` // 关联日志
}

// AlertReference 告警关联信息
type AlertReference struct {
    Page int                  `json:"page" bson:"page"`
    ID   primitive.ObjectID   `json:"id" bson:"id"`
}
```

## 4. API 请求/响应结构

### 4.1 通用响应

```go
// APIResponse 通用API响应
type APIResponse struct {
    Result  string      `json:"result,omitempty"`
    Message string      `json:"message,omitempty"`
    Data    interface{} `json:"data,omitempty"`
}

// PaginatedResponse 分页响应
type PaginatedResponse struct {
    Page     int         `json:"page"`
    LastPage int         `json:"last_page"`
    Data     interface{} `json:"data"`
}

// InfoResponse 系统信息响应
type InfoResponse struct {
    TimestampLastUpdate string `json:"timestamp_lastupdate"`
    CountAlert          int    `json:"count_alert"`
    TimestampRunningTime string `json:"timestamp_runningtime"`
}
```

### 4.2 PWN 下载请求

// PWNDownloadRequest PWN下载请求

## 5. 配置结构

```go
// Config 服务器配置
type Config struct {
    HTTPAddr   string `mapstructure:"http_addr"`
    HTTPPort   int    `mapstructure:"http_port"`
    TCPAddr    string `mapstructure:"tcp_addr"`
    TCPPort    int    `mapstructure:"tcp_port"`
    MongoDBURI string `mapstructure:"mongodb_uri"`
    Database   string `mapstructure:"database"`
    Token      string `mapstructure:"token"`
    
    // 插件配置
    PluginDir string `mapstructure:"plugin_dir"`
}

// DefaultConfig 默认配置
func DefaultConfig() *Config {
    return &Config{
        HTTPAddr:   "0.0.0.0",
        HTTPPort:   1337,
        TCPAddr:    "0.0.0.0",
        TCPPort:    8023,
        MongoDBURI: "mongodb://127.0.0.1:27017",
        Database:   "aoiawd",
        PluginDir:  "./plugins",
    }
}
```

## 6. 插件相关结构

```go
// Plugin 接口定义
type Plugin interface {
    Name() string
    Register(manager *Manager)
}

// HookFunc 钩子函数类型
type HookFunc func(data interface{}) interface{}

// PluginManager 插件管理器
type PluginManager struct {
    hooks   map[string][]HookFunc
    plugins []string
    caller  interface{}
    mu      sync.RWMutex
}

// PluginListResponse 插件列表响应
type PluginListResponse struct {
    Result  string   `json:"result"`
    Message string   `json:"message"`
    Data    []string `json:"data"`
}
```
