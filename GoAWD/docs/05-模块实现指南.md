## 1. Core Server 模块

### 1.1 入口点 `cmd/server/main.go`

```go

    "context"
    "flag"
    "log"
    "os"
    "os/signal"
    "syscall"
    
    "goawd/internal/config"
    "goawd/internal/core"

func main() {
    // 解析命令行参数
    httpAddr := flag.String("http", "0.0.0.0:1337", "HTTP server address")
    tcpAddr := flag.String("tcp", "0.0.0.0:8023", "TCP receiver address")
    mongoURI := flag.String("mongo", "mongodb://127.0.0.1:27017", "MongoDB URI")
    token := flag.String("token", "", "Access token (random if empty)")
    flag.Parse()
    
    // 配置
    cfg := &config.Config{
        HTTPAddr:   *httpAddr,
        TCPAddr:    *tcpAddr,
        MongoDBURI: *mongoURI,
        Token:      *token,
    }
    
    // 创建服务器
    server, err := core.NewServer(cfg)
    if err != nil {
        log.Fatalf("failed to create server: %v", err)
    }
    
    // 优雅关闭
    ctx, cancel := context.WithCancel(context.Background())
    defer cancel()
    
    go func() {
        sigCh := make(chan os.Signal, 1)
        signal.Notify(sigCh, syscall.SIGINT, syscall.SIGTERM)
        <-sigCh
        log.Println("shutting down...")
        cancel()
    }()
    
    // 启动服务器
    log.Printf("GoAWD starting...")
    log.Printf("HTTP: %s", *httpAddr)
    log.Printf("TCP:  %s", *tcpAddr)
    
    if err := server.Run(ctx); err != nil {
        log.Fatalf("server error: %v", err)
    }
}
```

## 2. TCP Receiver 模块

### 2.1 接收器实现 `internal/core/receiver.go`

```go
package core

import (
    "bufio"
    "context"
    "encoding/json"
    "log"
    "net"
    "sync"
    
    "goawd/internal/plugin"
    "goawd/internal/types"
)

type Receiver struct {
    port       int
    db         *Database
    pluginMgr  *plugin.Manager
    hub        *Hub
    
    // PWN 会话管理
    pwnSockets sync.Map // connID -> *types.PwnSocket
    pwnProcess sync.Map // pid -> *types.PwnProcess
    
    // 进程快照
    currentProcs sync.Map // pid -> *types.ProcessInfo
    
    listener   net.Listener
    mu         sync.RWMutex
}

func NewReceiver(port int, db *Database, hub *Hub, pluginMgr *plugin.Manager) *Receiver {
    return &Receiver{
        port:      port,
        db:        db,
        hub:       hub,
        pluginMgr: pluginMgr,
    }
}

func (r *Receiver) Start(ctx context.Context) error {
    var err error
    r.listener, err = net.Listen("tcp", fmt.Sprintf(":%d", r.port))
    if err != nil {
        return fmt.Errorf("failed to listen: %w", err)
    }
    defer r.listener.Close()
    
    log.Printf("TCP Receiver listening on :%d", r.port)
    
    for {
        conn, err := r.listener.Accept()
        if err != nil {
            select {
            case <-ctx.Done():
                return ctx.Err()
            default:
                log.Printf("accept error: %v", err)
                continue
            }
        }
        go r.handleConn(ctx, conn)
    }
}

func (r *Receiver) handleConn(ctx context.Context, conn net.Conn) {
    defer func() {
        if err := recover(); err != nil {
            log.Printf("panic in handleConn: %v", err)
        }
        conn.Close()
    }()
    
    connID := int(conn.(interface{ File() *os.File }).File().Fd())
    scanner := bufio.NewScanner(conn)
    
    for scanner.Scan() {
        select {
        case <-ctx.Done():
            return
        default:
            data := scanner.Bytes()
            r.processMessage(data, conn, connID)
        }
    }
    
    // 连接关闭，处理 PWN 流结束
    r.handlePWNClose(connID)
}

func (r *Receiver) processMessage(data []byte, conn net.Conn, connID int) {
    // 检查是否是 PWN 流模式
    if _, ok := r.pwnSockets.Load(connID); ok {
        r.processPWNStream(data, connID)
        return
    }
    
    // 解析 JSON 消息
    var msg types.ProbeMessage
    if err := json.Unmarshal(data, &msg); err != nil {
        log.Printf("json decode error: %v", err)
        return
    }
    
    switch msg.Type {
    case types.MsgTypeWeb:
        r.handleWeb(msg.Data, conn)
    case types.MsgTypeNewFile:
        r.handleNewFile(msg.Data)
    case types.MsgTypeNewProcess:
        r.handleNewProcess(msg.Data)
    case types.MsgTypeCurrentProcess:
        r.handleProcessList(msg.Data)
    case types.MsgTypePWN:
        r.handleNewPWN(msg.Data, conn, connID)
    case types.MsgTypePing:
        r.handlePing(conn)
    }
}
```

```go
func (r *Receiver) handleWeb(data interface{}, conn net.Conn) {
    // 转换为 WebLogData
    webData, ok := data.(*types.WebLogData)
    if !ok {
        log.Printf("invalid web data type")
        return
    }
    
    // URL 解码
    decoded := r.decodeWebData(webData)
    
    // 调用插件处理
    processed := r.pluginMgr.Invoke(nil, "Web", "processLog", decoded)
    
    // 发送响应（篡改后的 buffer）
    if processedData, ok := processed.(*types.WebLogData); ok {
        response := base64.StdEncoding.EncodeToString([]byte(processedData.Buffer))
        conn.Write([]byte(response + "\n"))
    }
    
    // 插入数据库
    r.db.InsertOne(context.Background(), "web", webData)
    
    // 广播更新
    r.hub.Notify("web")
}

func (r *Receiver) decodeWebData(data *types.WebLogData) *types.WebLogData {
    decoded := &types.WebLogData{
        Script: data.Script,
        Method: data.Method,
        URI:    data.URI,
        Remote: data.Remote,
        Header: make(map[string]string),
        GET:    make(map[string]string),
        POST:   make(map[string]string),
        Cookie: make(map[string]string),
        File:   data.File,
        Buffer: data.Buffer,
    }
    
    for k, v := range data.Header {
        decoded.Header[k] = url.QueryUnescape(v)
    }
    for k, v := range data.GET {
        decoded.GET[k] = url.QueryUnescape(v)
    }
    for k, v := range data.POST {
        decoded.POST[k] = url.QueryUnescape(v)
    }
    for k, v := range data.Cookie {
        decoded.Cookie[k] = url.QueryUnescape(v)
    }
    
    return decoded
}
```

```go
    if !ok {
        return
    }

    r.pwnProcess.Delete(socket.PID)
    
    // 广播更新
    r.hub.Notify("pwn")
```

## 3. HTTP Server 模块

### 3.1 服务器实现 `internal/core/server.go`

```go
package core

import (
    "context"
    "fmt"
    "log"
    "net/http"
    
    "github.com/go-chi/chi/v5"
    "github.com/go-chi/chi/v5/middleware"
    
    "goawd/internal/api"
    "goawd/internal/config"
    "goawd/internal/plugin"
)

type Server struct {
    config     *config.Config
    httpServer *http.Server
    receiver   *Receiver
    db         *Database
    hub        *Hub
    pluginMgr  *plugin.Manager
    router     *chi.Mux
}

func NewServer(cfg *config.Config) (*Server, error) {
    if err := cfg.Validate(); err != nil {
        return nil, err
    }
    
    db, err := NewDatabase(cfg.MongoDBURI, cfg.Database)
    if err != nil {
        return nil, fmt.Errorf("failed to connect to MongoDB: %w", err)
    }
    
    hub := NewHub()
    pluginMgr := plugin.NewManager()
    receiver := NewReceiver(parsePort(cfg.TCPAddr), db, hub, pluginMgr)
    
    s := &Server{
        config:    cfg,
        db:        db,
        hub:       hub,
        pluginMgr: pluginMgr,
        receiver:  receiver,
    }
    
    s.setupRouter()
    
    return s, nil
}

func (s *Server) setupRouter() {
    s.router = chi.NewRouter()
    
    // 中间件
    s.router.Use(middleware.Logger)
    s.router.Use(middleware.Recoverer)
    s.router.Use(s.corsMiddleware)
    
    // Token 鉴权中间件
    s.router.Use(s.tokenAuthMiddleware)
    
    // 健康检查
    s.router.Get("/api/ping", func(w http.ResponseWriter, r *http.Request) {
        w.Write([]byte("pong"))
    })
    
    // WebSocket
    s.router.Get("/websocket", s.hub.HandleWebSocket)
    
    // API v1 路由
    s.router.Route("/api/v1", func(r chi.Router) {
        apiV1 := api.NewV1(s.db, s.hub, s.pluginMgr, s.receiver, s.config.Token)
        r.Get("/info", apiV1.Info)
        r.Get("/listweb", apiV1.ListWeb)
        r.Get("/webdetail", apiV1.WebDetail)
        r.Get("/downloadwebautoscript", apiV1.DownloadWebAutoScript)
        r.Get("/listpwn", apiV1.ListPWN)
        r.Get("/pwndetail", apiV1.PWNDetail)
        r.Get("/downloadpwn", apiV1.DownloadPWN)
        r.Post("/generatepwnbin", apiV1.GeneratePWNBin)
        r.Get("/listfilesystem", apiV1.ListFilesystem)
        r.Get("/downloadfile", apiV1.DownloadFile)
        r.Get("/listprocess", apiV1.ListProcess)
        r.Get("/listcurrentprocess", apiV1.ListCurrentProcess)
        r.Get("/currentprocess", apiV1.CurrentProcess)
        r.Get("/listalert", apiV1.ListAlert)
        r.Get("/listplugin", apiV1.ListPlugin)
        r.Get("/reloadplugin", apiV1.ReloadPlugin)
    })
    
    // 静态文件
    s.router.Handle("/*", http.FileServer(http.Dir("public")))
}

func (s *Server) Run(ctx context.Context) error {
    // 启动 HTTP Server
    s.httpServer = &http.Server{
        Addr:    s.config.HTTPAddr,
        Handler: s.router,
    }
    
    go func() {
        log.Printf("HTTP Server listening on %s", s.config.HTTPAddr)
        if err := s.httpServer.ListenAndServe(); err != http.ErrServerClosed {
            log.Printf("HTTP server error: %v", err)
        }
    }()
    
    // 启动 TCP Receiver
    go func() {
        if err := s.receiver.Start(ctx); err != context.Canceled {
            log.Printf("TCP receiver error: %v", err)
        }
    }()
    
    // 启动 WebSocket Hub
    go s.hub.Run(ctx)
    
    // 加载插件
    s.pluginMgr.LoadPlugins(s.config.PluginDir)
    
    <-ctx.Done()
    
    // 优雅关闭
    shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
    defer cancel()
    
    return s.httpServer.Shutdown(shutdownCtx)
}
```

## 4. WebSocket Hub 模块

### 4.1 Hub 实现 `internal/core/websocket.go`

```go
package core

import (
    "context"
    "log"
    "net/http"
    "sync"
    "time"
    
    "github.com/gorilla/websocket"
)

var upgrader = websocket.Upgrader{
    CheckOrigin: func(r *http.Request) bool {
        return true
    },
}

type Client struct {
    conn *websocket.Conn
    send chan []byte
}

type Hub struct {
    clients    map[*Client]bool
    broadcast  chan *Message
    register   chan *Client
    unregister chan *Client
    
    // 延迟广播
    pending    map[string]bool
    pendingMu  sync.Mutex
}

type Message struct {
    Operation string `json:"operation"`
    Type      string `json:"type"`
}

func NewHub() *Hub {
    return &Hub{
        clients:    make(map[*Client]bool),
        broadcast:  make(chan *Message, 256),
        register:   make(chan *Client),
        unregister: make(chan *Client),
        pending:    make(map[string]bool),
    }
}

func (h *Hub) Run(ctx context.Context) {
    // 定时触发广播
    ticker := time.NewTicker(1500 * time.Millisecond)
    defer ticker.Stop()
    
    for {
        select {
        case <-ctx.Done():
            return
            
        case client := <-h.register:
            h.clients[client] = true
            log.Printf("WebSocket client connected, total: %d", len(h.clients))
            
        case client := <-h.unregister:
            if _, ok := h.clients[client]; ok {
                delete(h.clients, client)
                close(client.send)
                log.Printf("WebSocket client disconnected, total: %d", len(h.clients))
            }
            
        case msg := <-h.broadcast:
            for client := range h.clients {
                select {
                case client.send <- msg.Data:
                default:
                    close(client.send)
                    delete(h.clients, client)
                }
            }
            
        case <-ticker.C:
            h.triggerPendingBroadcast()
        }
    }
}

func (h *Hub) Notify(msgType string) {
    h.pendingMu.Lock()
    defer h.pendingMu.Unlock()
    h.pending[msgType] = true
}

func (h *Hub) triggerPendingBroadcast() {
    h.pendingMu.Lock()
    pending := h.pending
    h.pending = make(map[string]bool)
    h.pendingMu.Unlock()
    
    for msgType := range pending {
        msg := &Message{
            Operation: "reload",
            Type:      msgType,
        }
        data, _ := json.Marshal(msg)
        
        for client := range h.clients {
            select {
            case client.send <- data:
            default:
                close(client.send)
                delete(h.clients, client)
            }
        }
    }
}

func (h *Hub) HandleWebSocket(w http.ResponseWriter, r *http.Request) {
    conn, err := upgrader.Upgrade(w, r, nil)
    if err != nil {
        log.Printf("websocket upgrade error: %v", err)
        return
    }
    
    client := &Client{
        conn: conn,
        send: make(chan []byte, 256),
    }
    
    h.register <- client
    
    go client.writePump()
    go client.readPump(h)
}

func (c *Client) readPump(h *Hub) {
    defer func() {
        h.unregister <- c
        c.conn.Close()
    }()
    
    for {
        _, _, err := c.conn.ReadMessage()
        if err != nil {
            break
        }
    }
}

func (c *Client) writePump() {
    defer c.conn.Close()
    
    for msg := range c.send {
        if err := c.conn.WriteMessage(websocket.TextMessage, msg); err != nil {
            break
        }
    }
}
```

### 5.1 FlagBuster 插件 `plugins/flagbuster.go`

```go
package plugins

import (
    "regexp"
    "github.com/google/uuid"
    
    "goawd/internal/plugin"
)

type FlagBuster struct{}

func init() {
    plugin.RegisterPlugin(&FlagBuster{})
}

func (f *FlagBuster) Name() string {
    return "FlagBuster"
}

func (f *FlagBuster) Register(m *plugin.Manager) {
    m.Register("Web", "processLog", f.processWebBuffer)
}

var flagRegex = regexp.MustCompile(`\{"flag":"([^"]*)"`)

func (f *FlagBuster) processWebBuffer(data interface{}) interface{} {
    webData, ok := data.(map[string]interface{})
    if !ok {
        return data
    }
    
    buffer, ok := webData["buffer"].(string)
    if !ok {
        return data
    }
    
    fakeFlag := `{"flag":"` + uuid.New().String() + `"`
    
    if flagRegex.MatchString(buffer) {
        webData["buffer"] = flagRegex.ReplaceAllString(buffer, fakeFlag)
        
        // 获取插件管理器并设置告警
        mgr := plugin.GetManager()
        if invoker := mgr.GetCaller(); invoker != nil {
            if setter, ok := invoker.(interface{ SetAlert(string, string) }); ok {
                setter.SetAlert("FlagBuster", 
                    "发现本次Web应答包含flag字段，已被替换为: "+fakeFlag)
            }
        }
    }
    
    return data
}
```

## 6. RoundWorm 探针模块

### 6.1 入口点 `cmd/roundworm/main.go`

```go
package main

import (
    "context"
    "flag"
    "log"
    "os"
    "os/signal"
    "syscall"
    
    "goawd/pkg/inotify"
    "goawd/pkg/proc"
)

func main() {
    serverAddr := flag.String("s", "127.0.0.1", "Server address")
    serverPort := flag.Int("p", 8023, "Server port")
    watchDirs := flag.String("w", "/tmp", "Watch directories (separated by ;)")
    interval := flag.Duration("i", 100*time.Millisecond, "Process scan interval")
    daemon := flag.Bool("d", false, "Run as daemon")
    flag.Parse()
    
    if *daemon {
        // 守护进程化
        if os.Getppid() != 1 {
            syscall.Kill(os.Getppid(), syscall.SIGTERM)
            os.Exit(0)
        }
    }
    
    ctx, cancel := context.WithCancel(context.Background())
    defer cancel()
    
    go func() {
        sigCh := make(chan os.Signal, 1)
        signal.Notify(sigCh, syscall.SIGINT, syscall.SIGTERM)
        <-sigCh
        cancel()
    }()
    
    sender, err := NewTCPSender(*serverAddr, *serverPort)
    if err != nil {
        log.Fatalf("failed to connect to server: %v", err)
    }
    defer sender.Close()
    
    // 启动文件系统监控
    dirs := strings.Split(*watchDirs, ";")
    go inotify.Watch(dirs, sender)
    
    // 启动进程监控
    go proc.Scan(*interval, sender)
    
    <-ctx.Done()
    log.Println("RoundWorm shutting down")
}
```

## 7. Guardian 探针模块

### 7.1 入口点 `cmd/guardian/main.go`

```go
package main

import (
    "encoding/base64"
    "encoding/json"
    "flag"
    "fmt"
    "log"
    "net"
    "os"
    "os/exec"
    "path/filepath"
    
    "github.com/creack/pty"
)

func main() {
    binary := flag.String("binary", "", "PWN binary to run")
    serverHost := flag.String("host", "127.0.0.1", "Server host")
    serverPort := flag.Int("port", 8023, "Server port")
    flag.Parse()
    
    if *binary == "" {
        log.Fatal("binary is required")
    }
    
    // 启动子进程
    cmd := exec.Command(*binary)
    ptmx, err := pty.Start(cmd)
    if err != nil {
        log.Fatalf("failed to start pty: %v", err)
    }
    defer ptmx.Close()
    
    // 等待子进程启动
    time.Sleep(1 * time.Second)
    
    // 读取内存映射
    maps := readMemoryMap(cmd.Process.Pid)
    
    // 连接到服务器
    stdoutConn, err := net.Dial("tcp", fmt.Sprintf("%s:%d", *serverHost, *serverPort))
    if err != nil {
        log.Fatalf("failed to connect stdout: %v", err)
    }
    defer stdoutConn.Close()
    
    stdinConn, err := net.Dial("tcp", fmt.Sprintf("%s:%d", *serverHost, *serverPort))
    if err != nil {
        log.Fatalf("failed to connect stdin: %v", err)
    }
    defer stdinConn.Close()
    
    // 发送初始化信息
    exeName := filepath.Base(*binary)
    initMsg := map[string]interface{}{
        "type": "pwn",
        "data": map[string]interface{}{
            "file": exeName,
            "type": "stdout",
            "pid":  cmd.Process.Pid,
            "maps": maps,
        },
    }
    
    encoder := json.NewEncoder(stdoutConn)
    encoder.Encode(initMsg)
    
    initMsg["type"] = "stdin"
    encoder = json.NewEncoder(stdinConn)
    encoder.Encode(initMsg)
    
    // 双向转发
    done := make(chan struct{})
    
    go func() {
        io.Copy(stdoutConn, ptmx)
        done <- struct{}{}
    }()
    
    go func() {
        io.Copy(ptmx, stdinConn)
        done <- struct{}{}
    }()
    
    <-done
}

func readMemoryMap(pid int) string {
    path := fmt.Sprintf("/proc/%d/maps", pid)
    data, err := os.ReadFile(path)
    if err != nil {
        return ""
    }
    return base64.StdEncoding.EncodeToString(data)
}
```
